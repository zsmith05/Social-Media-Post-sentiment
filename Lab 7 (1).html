from tensorflow.keras.layers import TextVectorization
from tensorflow import keras
from sklearn.model_selection import StratifiedKFold
import numpy as np

text_column_name = 'tweet'
target_column_name = 'Sentiment'

# Extract text data, handling potential missing values by filling with an empty string
text_data = df[text_column_name].fillna('')

# Extract target labels
target_labels = df[target_column_name]

label_mapping = {
    -1: 0,          # Negative
    0: 1,           # Neutral
    1: 2,           # Positive
}
target_labels = target_labels.map(label_mapping)

# Ensure the Series is of integer type before passing to to_categorical
# This will raise an error if any values couldn't be mapped, indicating unhandled labels.
target_labels = target_labels.astype(int)

MAX_ART_LEN = 100
NUM_TOP_WORDS = 20000

# Use original labels for stratification, not the full dataframe
y_raw = df['Sentiment']  # Target variable (original labels for stratification)

# Determine the number of classes based on our mapping for to_categorical
num_classes = len(label_mapping)
y_ohe = keras.utils.to_categorical(target_labels, num_classes=num_classes)

stratified_kfold = StratifiedKFold(n_splits=10, shuffle=True, random_state=42)

# Correct usage: Call the split method and iterate to get train/test indices
# Then use these indices to split X and y_ohe
for train_index, test_index in stratified_kfold.split(text_data, y_raw):
    # Fit vectorizer only on training text to prevent data leakage
    vectorizer = TextVectorization(max_tokens=NUM_TOP_WORDS, output_sequence_length=MAX_ART_LEN)
    vectorizer.adapt(text_data.iloc[train_index])

    # Save as sequences with integers replacing words
    X_train = vectorizer(text_data.iloc[train_index]).numpy()
    X_test = vectorizer(text_data.iloc[test_index]).numpy()
    y_train_ohe = y_ohe[train_index]
    y_test_ohe = y_ohe[test_index]
    # For a simple train/test split, we can break after the first fold
    break

print('Shape of data tensor:', X_train.shape)
print('Shape of label tensor:', y_train_ohe.shape)

For dividing our test and training sets, we employed Stratified 10-Fold Cross-Validation. We selected this method specifically because our dataset exhibits class imbalance, meaning some categories appear far more frequently than others. Stratified k-fold validation ensures that each fold maintains the same class distribution as the original dataset, guaranteeing that minority classes are represented in both training and validation sets during each iteration. This prevents scenarios where a fold might be dominated by a single class, which would lead to poor model generalization and unreliable performance estimates. By cycling through 10 different train-validation splits, we obtain a better evaluation of model performance across the entire dataset rather than relying on a single random split, which could accidentally favor or disadvantage certain classes. This approach gives us greater confidence that our model has genuinely learned patterns from all classes, not just overfit to the majority classes.

For our tokenizing method, we used Keras Text Vectorization. This method converts raw text strings into discrete tokens and then maps these tokens into a numerical format that the model can process.

We set a maximum sequence length of 100 tokens because our dataset consists of social media posts, which are typicaly short. Social media platforms typically impose character limits and users tend to express ideas concisely, so the vast majority of our text samples are well under 100 tokens. This shorter max length is more appropriate than using a generic value like 512, as it reduces unnecessary padding, decreases memory consumption during training, and allows the model to focus on the actual content without processing excess zeros. Additionally, padding short sequences to an artificially high length can introduce noise and slow down training without providing meaningful signal to the model.

Evaluation Metric¶

We evaluated our model using the F1-score, which provides a balanced measure of model performance by combining both precision and recall into a single metric. The F1-score is particularly well-suited for our multi-class text classification task because it accounts for class imbalance in our dataset, which accuracy alone would fail to capture. By penalizing both false positives and false negatives equally, the F1-score ensures that the model performs well across all classes rather than simply optimizing for the majority class(class 1). We computed the weighted F1-score to account for the varying frequencies of each class, giving us a comprehensive view of overall model performance that reflects real-world prediction accuracy.

from tensorflow import keras
from tensorflow.keras import layers

def build_transformer(vocab_size, embed_dim, num_heads, dim_feedforward, dropout, max_len, num_classes):
    inputs = keras.Input(shape=(max_len,))

    # Embedding layer
    x = layers.Embedding(vocab_size, embed_dim)(inputs)

    # Positional encoding added to embeddings
    positions = keras.ops.arange(max_len)
    x = x + layers.Embedding(max_len, embed_dim)(positions)

    # Multi-head self attention
    attn_output = layers.MultiHeadAttention(num_heads=num_heads, key_dim=embed_dim // num_heads, dropout=dropout)(x, x)
    x = layers.LayerNormalization()(x + attn_output)

    # Feedforward block
    ff = layers.Dense(dim_feedforward, activation='relu')(x)
    ff = layers.Dropout(dropout)(ff)
    ff = layers.Dense(embed_dim)(ff)
    x = layers.LayerNormalization()(x + ff)

    # Mean pooling then classify
    x = layers.GlobalAveragePooling1D()(x)
    x = layers.Dropout(dropout)(x)
    outputs = layers.Dense(num_classes, activation='softmax')(x)

    return keras.Model(inputs, outputs)

from tensorflow.keras import layers
def build_cnn(vocab_size, embed_dim, num_filters, filter_sizes, num_classes, dropout=0.5, pre_trained_embeddings=None):
    inputs = keras.Input(shape=(MAX_ART_LEN,))

    # Embedding layer
    if pre_trained_embeddings is not None:
        x = layers.Embedding(vocab_size, embed_dim, weights=[pre_trained_embeddings], trainable=False)(inputs)  # Freeze embeddings
    else:
        x = layers.Embedding(vocab_size, embed_dim)(inputs)

    # Convolutional layers for each filter size
    pooled_outputs = []
    for fs in filter_sizes:
        # Apply convolution + ReLU + max pooling for each filter size
        conv = layers.Conv1D(num_filters, fs, activation='relu')(x)
        pool = layers.GlobalMaxPooling1D()(conv)
        pooled_outputs.append(pool)

    # Concatenate pooled outputs and pass through fully connected layer
    cat = layers.Concatenate()(pooled_outputs)
    cat = layers.Dropout(dropout)(cat)
    outputs = layers.Dense(num_classes, activation='softmax')(cat)

    return keras.Model(inputs, outputs)

The Default CNN uses 100 filters with a filter size of [3,4,5], capturing tri-grams, four-grams, and five-grams respectively. Our adjusted CNN uses 200 filters instead, with a filter size of [3,5,7] which captures longer n-grams. These changes may benefit the model with its ability to capture more intricate relationships at the cost of higher computation and overfitting

The Default Transformer model has an embedded size of 128 with 4 attention heads and a feedforward dimension of 256. It's adjusted couterpart doubles the attention heads and feedforward dimension, exploring the trade-offs between model complexibility, and computational cost.

def macro_f1(y_true, y_pred):
    # Convert probabilities to one-hot via argmax instead of rounding
    y_pred_one_hot = tf.one_hot(tf.argmax(y_pred, axis=1), depth=NUM_CLASSES)
    y_true = tf.cast(y_true, tf.float32)

    f1_scores = []
    for i in range(NUM_CLASSES):
        tp = tf.reduce_sum(y_true[:, i] * y_pred_one_hot[:, i])
        fp = tf.reduce_sum((1 - y_true[:, i]) * y_pred_one_hot[:, i])
        fn = tf.reduce_sum(y_true[:, i] * (1 - y_pred_one_hot[:, i]))
        f1 = 2 * tp / (2 * tp + fp + fn + 1e-8)
        f1_scores.append(f1)
    return tf.reduce_mean(f1_scores)

def build_transformer_two_layers(vocab_size, embed_dim, num_heads, dim_feedforward, dropout, max_len, num_classes):
    inputs = keras.Input(shape=(max_len,))

    # Embedding layer
    x = layers.Embedding(vocab_size, embed_dim)(inputs)

    # Positional encoding added to embeddings
    positions = keras.ops.arange(max_len)
    x = x + layers.Embedding(max_len, embed_dim)(positions)

    # First multi-head self attention layer
    attn_output = layers.MultiHeadAttention(num_heads=num_heads, key_dim=embed_dim // num_heads, dropout=dropout)(x, x)
    x = layers.LayerNormalization()(x + attn_output)
    ff = layers.Dense(dim_feedforward, activation='relu')(x)
    ff = layers.Dropout(dropout)(ff)
    ff = layers.Dense(embed_dim)(ff)
    x = layers.LayerNormalization()(x + ff)

    # Second multi-head self attention layer - takes output of first as input
    attn_output = layers.MultiHeadAttention(num_heads=num_heads, key_dim=embed_dim // num_heads, dropout=dropout)(x, x)
    x = layers.LayerNormalization()(x + attn_output)
    ff = layers.Dense(dim_feedforward, activation='relu')(x)
    ff = layers.Dropout(dropout)(ff)
    ff = layers.Dense(embed_dim)(ff)
    x = layers.LayerNormalization()(x + ff)

    # Mean pooling then classify
    x = layers.GlobalAveragePooling1D()(x)
    x = layers.Dropout(dropout)(x)
    outputs = layers.Dense(num_classes, activation='softmax')(x)

    return keras.Model(inputs, outputs)

Exceptional Work¶

For exceptional work, we added a Hybrid CNN-Transformer. This model uses a shared embedding layer whose output is fed simultaneously into a CNN branch and a Transformer branch. The CNN applies multi scale 1D convolutions, while the Transformer applies multi-head self attention with global average pooling. Both outputs are then concatenated and passed through a dropout layer for final classification

def build_hybrid(vocab_size, embed_dim, cnn_filters, cnn_kernel_sizes, num_heads, dim_feedforward, num_classes, dropout, max_len):
    inputs = keras.Input(shape=(max_len,))

    # Embedding layer
    x = layers.Embedding(vocab_size, embed_dim)(inputs)

    # Positional encoding added to embeddings
    positions = keras.ops.arange(max_len)
    x = x + layers.Embedding(max_len, embed_dim)(positions)

    # CNN feature extraction
    cnn_outputs = []
    for k in cnn_kernel_sizes:
        conv = layers.Conv1D(cnn_filters, k, activation='relu')(x)
        pool = layers.GlobalMaxPooling1D()(conv)
        cnn_outputs.append(pool)
    x_cnn = layers.Concatenate()(cnn_outputs)

    # Transformer processing
    attn_output = layers.MultiHeadAttention(num_heads=num_heads, key_dim=embed_dim // num_heads, dropout=dropout)(x, x)
    x_trans = layers.LayerNormalization()(x + attn_output)
    x_trans = layers.GlobalAveragePooling1D()(x_trans)

    # Combine CNN and Transformer features
    x_combined = layers.Concatenate()([x_cnn, x_trans])
    x_combined = layers.Dropout(dropout)(x_combined)
    outputs = layers.Dense(num_classes, activation='softmax')(x_combined)

    return keras.Model(inputs, outputs)
